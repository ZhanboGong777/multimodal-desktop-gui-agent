def method(tree):
    klass = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == "SemanticDecomposer")
    return next(n for n in klass.body if isinstance(n, ast.FunctionDef) and n.name == "_validate")


def boolean(test):
    if isinstance(test, ast.Compare) and ast.unparse(test) in labels:
        return ast.Name(id=labels[ast.unparse(test)], ctx=ast.Load())
    if isinstance(test, ast.BoolOp) and isinstance(test.op, ast.Or):
        children = [boolean(value) for value in test.values]
        if all(child is not None for child in children):
            return ast.BoolOp(op=ast.Or(), values=children)
    return None


conditions, memberships = [], []
for tree in trees:
    expressions, terms = [], []
    for node in ast.walk(method(tree)):
        if not isinstance(node, ast.If):
            continue
        converted = boolean(node.test)
        if converted is None:
            continue
        assert not node.orelse
        assert isinstance(node.body[-1], ast.Raise) and isinstance(node.body[-1].exc, ast.Call)
        assert isinstance(node.body[-1].exc.func, ast.Name) and node.body[-1].exc.func.id == "ValueError"
        for before_raise in node.body[:-1]:
            assert ast.dump(before_raise, include_attributes=False) == ast.dump(ast.parse(
                "prior = check_owners[signature[1:]]").body[0], include_attributes=False)
        assert all(isinstance(call.func, ast.Name) and call.func.id == "ValueError"
                   for call in ast.walk(node) if isinstance(call, ast.Call))
        expressions.append(compile(ast.fix_missing_locations(ast.Expression(converted)), "<guard>", "eval"))
        terms += [labels[ast.unparse(n)] for n in ast.walk(node.test) if isinstance(n, ast.Compare)]
    conditions.append(expressions)
    memberships.append(sorted(terms))
assert memberships[0] == memberships[1] == sorted(labels.values())
truth = []
for bits in itertools.product((False, True), repeat=6):
    env = dict(zip(sorted(labels.values()), bits))
    values = [any(eval(code, {"__builtins__": {}}, env) for code in clauses) for clauses in conditions]
    assert values[0] == values[1]
    truth.append({"conditions": env, "before_rejects": values[0], "after_rejects": values[1]})


class StripDiagnosticChange(ast.NodeTransformer):
    def visit_If(self, node):
        if boolean(node.test) is not None:
            return None
        return self.generic_visit(node)

    def visit_AnnAssign(self, node):
        if isinstance(node.target, ast.Name) and node.target.id == "check_owners":
            assert isinstance(node.value, ast.Dict) and not node.value.keys
            return None
        return self.generic_visit(node)

    def visit_Assign(self, node):
        if any(isinstance(t, ast.Subscript) and isinstance(t.value, ast.Name)
               and t.value.id == "check_owners" for t in node.targets):
            assert ast.unparse(node) == "check_owners[signature[1:]] = goal.goal_id"
            return None
        return self.generic_visit(node)


stripped = [StripDiagnosticChange().visit(copy.deepcopy(method(tree))) for tree in trees]
assert ast.dump(stripped[0], include_attributes=False) == ast.dump(stripped[1], include_attributes=False)


def outside_prompt_and_validate(tree):
    tree = copy.deepcopy(tree)
    for node in tree.body:
        if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == "SEMANTIC_SYSTEM_PROMPT" for t in node.targets):
            node.value = ast.Constant("<reviewed-prompt>")
    method(tree).body = [ast.Pass()]
    return ast.dump(tree, include_attributes=False)
