"""One migrated context node and Ruff with actual subprocess waits and pins."""
from __future__ import annotations

import ast
import hashlib
import json
import os
import subprocess
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

import psutil

ROOT = Path("C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree")
MIGRATION = ROOT / "Document/Week6/results/full_requirements_20261028/target_bindings_001/legacy_context_assertion_migration_001"
OUT = MIGRATION / "quickcheck_001"
RELATIVE = "tests/test_runtime_runner.py"
FUNCTION = "test_non_message_planning_context_remains_the_complete_observation"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def identity(process):
    return {"pid": process.pid, "create_time": process.create_time(), "exe": process.exe()}


def main():
    OUT.mkdir(exist_ok=False)
    migration = json.loads((MIGRATION / "migration.json").read_text("utf8"))
    paths = migration["code_pins_after"]
    before = {name: sha(ROOT / name) for name in paths}
    req_before = sha(ROOT / "requirements-dev.txt")
    assert before == paths and len(before) == 270 and req_before == migration["requirements_dev_sha256"]
    old_tree = ast.parse((MIGRATION / "source_before" / (RELATIVE + ".txt")).read_text("utf8"))
    new_tree = ast.parse((ROOT / RELATIVE).read_text("utf8"))
    old_fn = next(node for node in old_tree.body if isinstance(node, ast.FunctionDef) and node.name == FUNCTION)
    new_fn = next(node for node in new_tree.body if isinstance(node, ast.FunctionDef) and node.name == FUNCTION)
    new_statements = new_fn.body[len(old_fn.body):]
    assert len(new_statements) == 6 and isinstance(new_statements[0], ast.Assign)
    assert all(isinstance(node, ast.Assert) for node in new_statements[1:])
    del new_fn.body[len(old_fn.body):]
    assertion = new_fn.body[4]
    assert isinstance(assertion, ast.Assert) and isinstance(assertion.test.comparators[0], ast.Set)
    values = assertion.test.comparators[0].elts
    added = [index for index, item in enumerate(values) if isinstance(item, ast.Constant) and item.value == "target_bindings"]
    assert len(added) == 1
    values.pop(added[0])
    assert ast.dump(old_tree) == ast.dump(new_tree), "An existing AST statement changed beyond the one context key"
    receipts = []
    commands = [("pytest", [sys.executable, "-m", "pytest", "-q", RELATIVE + "::" + FUNCTION,
                             "--basetemp=" + str(OUT / "pytest_tmp"), "--tb=short"]),
                ("ruff", [sys.executable, "-m", "ruff", "check", RELATIVE])]
    for label, command in commands:
        started = datetime.now(timezone.utc).isoformat()
        timer = time.monotonic()
        descendants = {}
        with (OUT / f"{label}.stdout.txt").open("xb") as stdout, (OUT / f"{label}.stderr.txt").open("xb") as stderr:
            process = subprocess.Popen(command, cwd=ROOT, stdout=stdout, stderr=stderr,
                                       creationflags=subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0)
            launcher = identity(psutil.Process(process.pid))
            while process.poll() is None:
                try:
                    for child in psutil.Process(process.pid).children(recursive=True):
                        try:
                            item = identity(child)
                            descendants[(item["pid"], item["create_time"])] = item
                        except psutil.Error:
                            pass
                except psutil.Error:
                    pass
                time.sleep(.05)
            actual_wait = process.wait()
        receipt = {"command": command, "launcher": launcher, "observed_descendants": list(descendants.values()),
                   "started_utc": started, "finished_utc": datetime.now(timezone.utc).isoformat(),
                   "elapsed_seconds": time.monotonic() - timer, "actual_wait_exit_code": actual_wait,
                   "stdout_sha256": sha(OUT / f"{label}.stdout.txt"), "stderr_sha256": sha(OUT / f"{label}.stderr.txt")}
        (OUT / f"{label}.receipt.json").write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + "\n", encoding="utf8")
        receipts.append(receipt)
    after = {name: sha(ROOT / name) for name in paths}
    req_after = sha(ROOT / "requirements-dev.txt")
    assert after == before and req_after == req_before
    validation = {"schema": "week6.target_binding.context_migration.quickcheck.v1",
                  "scope": "One migrated node and owned Ruff only; not a full or affected regression",
                  "receipts": receipts, "code_pin_count": len(before), "code_pins_before": before,
                  "code_pins_after": after, "code_pins_unchanged": True,
                  "requirements_dev_sha256": req_after, "requirements_dev_unchanged": True,
                  "ast_only_one_context_key_and_five_new_binding_assertions": True,
                  "all_other_existing_test_ast_equal": True, "test_count_unchanged": True,
                  "original_affected001_exit1_not_reclassified": True,
                  "checks_passed": all(item["actual_wait_exit_code"] == 0 for item in receipts),
                  "no_production_auditor_other_test_edits": True, "model_gui_calls": 0}
    (OUT / "validation.json").write_text(json.dumps(validation, ensure_ascii=False, indent=2) + "\n", encoding="utf8")
    for name in ["archive_and_migrate_001.py", "quickcheck_001.py"]:
        with (MIGRATION / (name + ".txt")).open("xb") as handle:
            handle.write(Path(__file__).with_name(name).read_bytes())
    print(json.dumps({"validation": str(OUT / "validation.json"), "sha256": sha(OUT / "validation.json"),
                      "test_sha256": after[RELATIVE], "actual_exits": [item["actual_wait_exit_code"] for item in receipts]}, ensure_ascii=False))
    assert all(item["actual_wait_exit_code"] == 0 for item in receipts)


if __name__ == "__main__":
    main()
