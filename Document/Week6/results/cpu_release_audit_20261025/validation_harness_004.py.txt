from datetime import UTC, datetime
from pathlib import Path
import hashlib
import json
import os
import subprocess
import time

ROOT = Path('C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree')
OUT = ROOT / 'Document/Week6/results/release_validation_completed_20261024_004'
OUT.mkdir(parents=True, exist_ok=False)
PYTHON = 'D:/Developer/multimodal-desktop-gui-agent/.venv-week5/Scripts/python.exe'

def pins():
    paths = [p for name in ('src', 'tests', 'scripts', 'configs')
             for p in (ROOT / name).rglob('*') if p.is_file() and p.suffix in ('.py', '.json', '.yaml')]
    paths += [ROOT / 'pyproject.toml', ROOT / 'README.md']
    return {p.relative_to(ROOT).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(paths)}

before = pins()
env = os.environ.copy()
env.update(PYTHONUTF8='1', PYTHONIOENCODING='utf-8')
command = [PYTHON, '-m', 'pytest', '-q', '--basetemp=.pytest-tmp-week6-completed004', '--tb=short']
start = datetime.now(UTC).isoformat()
tick = time.perf_counter()
with (OUT / 'pytest_stdout.txt').open('w', encoding='utf-8') as log:
    process = subprocess.run(command, cwd=ROOT, env=env, stdout=log, stderr=subprocess.STDOUT, check=False)
after = pins()
result = {'schema': 'week6.release.cpu_validation.v1', 'started_at_utc': start,
          'finished_at_utc': datetime.now(UTC).isoformat(), 'command': command,
          'environment_overrides': {'PYTHONUTF8': '1', 'PYTHONIOENCODING': 'utf-8'},
          'exit_code': process.returncode, 'elapsed_seconds': time.perf_counter()-tick,
          'tested_source_pins_before': before, 'tested_sources_unchanged': before == after,
          'changed_tested_sources': [p for p in before.keys() | after.keys() if before.get(p) != after.get(p)],
          'stdout': 'pytest_stdout.txt'}
(OUT / 'validation.json').write_text(json.dumps(result, ensure_ascii=False, indent=2)+'\n', encoding='utf-8')
print(json.dumps({k: result[k] for k in ('exit_code', 'elapsed_seconds', 'tested_sources_unchanged')}))
print((OUT / 'pytest_stdout.txt').read_text(encoding='utf-8')[-5000:])
raise SystemExit(process.returncode)
