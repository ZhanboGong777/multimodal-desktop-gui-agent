"""Bind the finished independent review without touching the code domain."""
from __future__ import annotations

import ast
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path("C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree")
BASE = ROOT / "Document/Week6/results/full_requirements_20261028"
OUT = BASE / "review/semantic_repair_feedback_review_001"
PRIVATE = Path(__file__).parent


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def bound(path):
    return {"path": str(path), "sha256": sha(path), "bytes": path.stat().st_size}


def load(path):
    return json.loads(path.read_text(encoding="utf8"))


def node(tree, kind, name):
    return next(n for n in tree.body if isinstance(n, kind) and n.name == name)


def main():
    snapshot = OUT / "source_after_001"
    snapshot.mkdir(exist_ok=False)
    copies = []
    for relative in ("scripts/week6_audit_semantic_acceptance.py", "tests/test_week6_semantic_repair_feedback_review.py"):
        source = ROOT / relative
        dest = snapshot / (relative + ".txt")
        dest.parent.mkdir(parents=True)
        with dest.open("xb") as stream:
            stream.write(source.read_bytes())
        assert sha(source) == sha(dest)
        copies.append({"source": bound(source), "destination": bound(dest)})
    for name in ("run_repair_feedback_review_001.py", "run_repair_feedback_review_002.py", "run_repair_feedback_review_003.py", Path(__file__).name):
        source = PRIVATE / name
        dest = snapshot / (name + ".txt")
        with dest.open("xb") as stream:
            stream.write(source.read_bytes())
        assert sha(source) == sha(dest)
        copies.append({"source": bound(source), "destination": bound(dest)})
    old_source = BASE / "live/normal_011/run/source_snapshot/src/gui_agent/planning/semantic_workflow.py.txt"
    current_source = ROOT / "src/gui_agent/planning/semantic_workflow.py"
    old_tree = ast.parse(old_source.read_text(encoding="utf8"))
    current_tree = ast.parse(current_source.read_text(encoding="utf8"))

    def assigned(tree, name):
        return next(n.value for n in tree.body if isinstance(n, ast.Assign)
                    and any(isinstance(t, ast.Name) and t.id == name for t in n.targets))

    assert ast.dump(assigned(old_tree, "SEMANTIC_SYSTEM_PROMPT")) == ast.dump(assigned(current_tree, "SEMANTIC_SYSTEM_PROMPT"))
    assert ast.dump(assigned(old_tree, "SEMANTIC_PROTOCOL_VERSION")) == ast.dump(assigned(current_tree, "SEMANTIC_PROTOCOL_VERSION"))
    for name in ("_caller_contract_issue", "_semantic_response_format", "_validate_native_predicates", "_goal_task", "_final_task"):
        assert ast.dump(node(old_tree, ast.FunctionDef, name)) == ast.dump(node(current_tree, ast.FunctionDef, name))
    for name in ("SemanticGoal", "SemanticPlan", "SemanticPlanResult", "SemanticWorkflowRunner", "_SemanticChild"):
        assert ast.dump(node(old_tree, ast.ClassDef, name)) == ast.dump(node(current_tree, ast.ClassDef, name))
    old_cls = node(old_tree, ast.ClassDef, "SemanticDecomposer")
    current_cls = node(current_tree, ast.ClassDef, "SemanticDecomposer")
    old_validate = next(n for n in old_cls.body if isinstance(n, ast.FunctionDef) and n.name == "_validate")
    current_validate = next(n for n in current_cls.body if isinstance(n, ast.FunctionDef) and n.name == "_validate")

    class NormalizeErrorType(ast.NodeTransformer):
        def visit_Name(self, value):
            if value.id == "_DuplicateSemanticPostconditionsError":
                value.id = "ValueError"
            return value

    assert ast.dump(old_validate) == ast.dump(NormalizeErrorType().visit(current_validate))
    pin_manifest = load(BASE / "grounding_consistency_001/final_source_freeze_001.json")
    changed = [relative for relative, expected in pin_manifest["code_pins"].items() if sha(ROOT / relative) != expected]
    assert sorted(changed) == ["scripts/week6_audit_semantic_acceptance.py", "src/gui_agent/planning/semantic_workflow.py"]
    final_run = load(OUT / "cpu_validation_003/audit.json")
    assert final_run["all_narrow_checks_passed"] and final_run["focused_source_pins_unchanged"]
    assert final_run["summary"] == "19 passed in 5.73s"
    audit = {
        "schema": "week6.independent.semantic_repair_boundary_review.v1",
        "reviewed_utc": datetime.now(timezone.utc).isoformat(),
        "reviewer": "/root/week6_replanning_review (independent Codex agent)",
        "frozen_for_root_unified_validation": True,
        "actual_own_target": bound(OUT / "cpu_validation_003/audit.json"),
        "source_before": bound(OUT / "source_before_manifest_001.json"),
        "initial_narrow_failures_preserved": [bound(OUT / f"cpu_validation_00{i}/audit.json") for i in (1, 2)],
        "fixture_assertion_corrections": [bound(OUT / f"assertion_corrections_00{i}.json") for i in (1, 2)],
        "source_scope": {"old_native_protocol_and_original_system_same_AST": True,
                         "raw_predicate_and_schema_and_caller_preflight_and_final_goal_compilers_same_AST": True,
                         "goal_plan_result_and_execution_classes_same_AST": True,
                         "validate_same_AST_after_only_normalizing_typed_exception_to_previous_ValueError": True,
                         "changed_preexisting_f4_freeze_paths": changed,
                         "production_source": bound(current_source)},
        "independent_behavior_checks": [
            "Ordinary ValueError with identical duplicate-message text does not receive typed feedback",
            "Duplicate then JSON failure removes old hint; hints do not accumulate",
            "Three identical bad replies remain rejected with zero child/input/completion",
            "Native IDs/instructions/checks/reason and caller canaries never enter fixed system feedback",
            "Refused and needs_clarification remain native terminal outcomes",
            "Actual model-request and shared planning/deadline accounting remain bounded; late original reply is preserved only as transport evidence and cannot authorize input",
            "Actual SDK MockTransport body and saved requests have exact base/hint/schema ordering and unchanged output schema",
            "Original raw bad reply and model-owned repaired native map exactly to compiled goals; no local merge",
            "Auditor replays raw prior reply through frozen validators, compares exact error, and derives typed category locally",
            "Auditor rejects missing/tampered policy, extra/duplicate/misplaced system text, first-attempt hint, and a forged prior rejection for a locally valid native reply",
        ],
        "audit_enforcement": "Exact system equality, not prefix matching; exact four-key repair policy; original repair references and content lengths retained; current transport message binding remains unchanged",
        "limitations": ["No model, GUI, fixture or owner rerun", "Offline doubles and MockTransport do not prove a real model repairs its answer", "Narrow 19-case pass is not full regression or live semantic completion", "normal001 through normal011 remain failed historical attempts"],
        "copies": copies,
    }
    path = OUT / "independent_boundary_review_001.json"
    with path.open("x", encoding="utf8") as stream:
        json.dump(audit, stream, ensure_ascii=False, indent=2)
        stream.write("\n")
    print(json.dumps({"review": str(path), "sha256": sha(path), "copied_sources": len(copies)}, ensure_ascii=False))


if __name__ == "__main__":
    main()
