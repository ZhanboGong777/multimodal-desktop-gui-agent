class FreezeError(ValueError):
    pass


def require(condition, detail):
    if not condition:
        raise FreezeError(detail)


def sha(path):
    return sha256(path.read_bytes()).hexdigest()


def unique(pairs):
    value = {}
    for key, item in pairs:
        require(key not in value, 'duplicate JSON key')
        value[key] = item
    return value


def load(path):
    return json.loads(path.read_bytes(), object_pairs_hook=unique,
                      parse_constant=lambda _v: (_ for _ in ()).throw(FreezeError('nonfinite JSON')))


def bound(path, expected):
    require(type(expected) is str and re.fullmatch('[0-9a-f]{64}', expected), 'TODO: supply the actual 64-character proof SHA')
    require(path.is_file(), f'TODO: original evidence not supplied: {path}')
    require(sha(path) == expected, f'changed evidence: {path}')
    return load(path)


def write(path, value):
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        stream.write(json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False)+'\n')


def inventory():
    paths = {REPO/'pyproject.toml'}
    for folder in ('src', 'tests', 'scripts'):
        paths.update((REPO/folder).rglob('*.py'))
    paths.update(path for path in (REPO/'configs').rglob('*') if path.is_file())
    return {path.relative_to(REPO).as_posix(): sha(path) for path in sorted(paths)}


def ast_value(node):
    return ast.dump(node, include_attributes=False)


def check_legacy_test(old, current, proof, pins, prior):
    require(proof.get('schema') == 'week6.visual_guidance.legacy_test_migration.v1', 'TODO: unsupported migration proof schema')
    require(proof.get('path') == 'tests/test_runtime_runner.py'
            and proof.get('old_sha256') == prior['code_pins']['tests/test_runtime_runner.py']
            and proof.get('new_sha256') == pins['tests/test_runtime_runner.py']
            and proof.get('changed_function') == TEST_FUNCTION
            and proof.get('original_context_set_added_keys') == ['targeting_policy']
            and proof.get('original_assertions_preserved') is True
            and proof.get('original_function_AST_equal_after_only_added_import_key_policy_assertion_removed') is True
            and proof.get('every_other_test_module_AST_equal') is True
            and proof.get('old_test_count_preserved') is True, 'new migration proof is not bound to this test change')
    before, after = ast.parse(old.read_text('utf-8')), ast.parse(current.read_text('utf-8'))
    original = next(n for n in before.body if isinstance(n, ast.FunctionDef) and n.name == TEST_FUNCTION)
    changed = next(n for n in after.body if isinstance(n, ast.FunctionDef) and n.name == TEST_FUNCTION)
    expected_assert = ast.parse("assert context['targeting_policy'] == {'schema_version': GENERATION_TARGETING_POLICY_VERSION, 'instruction': GENERATION_TARGETING_POLICY_TEXT, 'generation_only': True}").body[0]
    counts = {'import': 0, 'key': 0, 'policy_assert': 0}
    kept = []
    for statement in changed.body:
        if (isinstance(statement, ast.ImportFrom) and statement.module == 'gui_agent.planning.target_bindings'
                and statement.level == 0
                and [(x.name, x.asname) for x in statement.names] == [
                    ('GENERATION_TARGETING_POLICY_TEXT', None), ('GENERATION_TARGETING_POLICY_VERSION', None)]):
            counts['import'] += 1
            continue
        if ast_value(statement) == ast_value(expected_assert):
            counts['policy_assert'] += 1
            continue
        for node in ast.walk(statement):
            if isinstance(node, ast.Set):
                matches = [x for x in node.elts if isinstance(x, ast.Constant) and x.value == 'targeting_policy']
                if matches:
                    require(len(matches) == 1, 'duplicate migrated context key')
                    node.elts.remove(matches[0]); counts['key'] += 1
        kept.append(statement)
    changed.body = kept
    require(counts == {'import': 1, 'key': 1, 'policy_assert': 1}
            and ast_value(changed) == ast_value(original)
            and ast_value(after) == ast_value(before), 'legacy test changed beyond the one context-key migration and its policy assertion')
    return counts


def check_source_scope(prior, pins, proof):
    changed = {name for name in set(prior['code_pins']) | set(pins)
               if prior['code_pins'].get(name) != pins.get(name)}
    require(changed == CHANGED_PATHS, f'unexpected changed paths: {sorted(changed)}')
    require(proof.get('prior_freeze_sha256') == PRIOR_SHA
            and proof.get('source_before_manifest_sha256') == BEFORE_SHA
            and set(proof.get('changed_paths', [])) == CHANGED_PATHS
            and set(proof.get('runner_changed_methods', [])) == RUNNER_METHODS,
            'TODO: final source_delta_proof does not bind the authorised six paths and methods')
    old_helper = ast.parse((CHANGE/'source_before/code/src/gui_agent/planning/target_bindings.py').read_text('utf-8'))
    helper_path = REPO/'src/gui_agent/planning/target_bindings.py'
    new_helper = ast.parse(helper_path.read_text('utf-8'))
    allowed = {'GENERATION_TARGETING_POLICY_TEXT', 'GENERATION_TARGETING_POLICY_VERSION'}
    declarations, stripped = set(), []
    for node in new_helper.body:
        if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name) and node.targets[0].id in allowed:
            declarations.add(node.targets[0].id)
        elif isinstance(node, ast.FunctionDef) and node.name == 'generation_targeting_policy':
            declarations.add(node.name)
            statements = node.body[1:] if node.body and isinstance(node.body[0], ast.Expr) and isinstance(node.body[0].value, ast.Constant) else node.body
            expected = ast.parse("return {'schema_version': GENERATION_TARGETING_POLICY_VERSION, 'instruction': GENERATION_TARGETING_POLICY_TEXT, 'generation_only': True}").body
            require(len(statements) == 1 and ast_value(statements[0]) == ast_value(expected[0]), 'policy API is not the declared fresh pure dictionary')
        else:
            stripped.append(node)
    require(declarations == allowed | {'generation_targeting_policy'}, 'new targeting-policy declarations missing')
    reduced = deepcopy(new_helper); reduced.body = stripped
    require(ast_value(reduced) == ast_value(old_helper), 'old binding packet/schema/helper AST changed')
    old_runner = ast.parse((CHANGE/'source_before/code/src/gui_agent/runtime/runner.py').read_text('utf-8'))
    new_runner = ast.parse((REPO/'src/gui_agent/runtime/runner.py').read_text('utf-8'))
    old_class = next(n for n in old_runner.body if isinstance(n, ast.ClassDef) and n.name == 'TaskRunner')
    new_class = next(n for n in new_runner.body if isinstance(n, ast.ClassDef) and n.name == 'TaskRunner')
    old_methods = {n.name: n for n in old_class.body if isinstance(n, ast.FunctionDef)}
    observed = set()
    for index, method in enumerate(new_class.body):
        if isinstance(method, ast.FunctionDef) and method.name in RUNNER_METHODS:
            require(method.name in old_methods, 'new runner method was not authorised')
            if ast_value(method) != ast_value(old_methods[method.name]): observed.add(method.name)
            new_class.body[index] = old_methods[method.name]
    require(observed == RUNNER_METHODS and ast_value(new_runner) == ast_value(old_runner), 'runner changed beyond context and recovery context')
    # This is the actual pure helper API, not a guessed copy of its text.
    spec = importlib.util.spec_from_file_location('week6_frozen_visual_targeting_helper', helper_path)
    module = importlib.util.module_from_spec(spec)
    require(spec.loader is not None, 'cannot load the pure targeting helper')
    spec.loader.exec_module(module)
    policy, fresh_policy = module.generation_targeting_policy(), module.generation_targeting_policy()
    require(type(policy) is dict and set(policy) == {'schema_version', 'instruction', 'generation_only'}
            and policy == fresh_policy and policy is not fresh_policy
            and policy['schema_version'] == POLICY_VERSION and policy['generation_only'] is True
            and type(policy['instruction']) is str and policy['instruction'].strip()
            and policy == proof.get('generation_targeting_policy'), 'actual fixed targeting-policy API and proof disagree')
    return sorted(changed), policy

            and len(manifest['files']) == 512, 'unexpected original preservation manifest')
    seen, code = set(), {}
    for record in manifest['files']:
        destination = (CHANGE/record['destination']).resolve()
        require(destination.is_relative_to((CHANGE/'source_before').resolve()) and destination not in seen,
                'preservation path escapes or repeats')

            require(sha(Path(record['source'])) == record['sha256'], 'ended normal016 or negative audit original changed')
    require(code == prior['code_pins'], 'archived code does not match the previous freeze')
