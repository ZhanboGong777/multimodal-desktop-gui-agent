"""Synthetic dictionary boundaries only: no windows, inputs, models or task proof."""
from copy import deepcopy
import unittest

import audit_input_receipt_protocol as audit

EXPECTED = {"app": "synthetic-process", "id": 42, "title": "Synthetic protocol record"}


def records():
    result = []
    for index, second in enumerate((1, 3), 1):
        stamp = lambda fraction: f"2026-10-29T00:00:0{second}.{fraction:03d}Z"
        action = {"action_type": "click", "x": 7, "y": 11, "click_count": 1}
        request = {"id": f"input_{index:03d}", "action": action,
                   "required_window_title": EXPECTED["title"], "created_utc": stamp(0)}
        receipt = {"id": request["id"], "action": deepcopy(action), "success": True,
                   "transport": "Sky click", "timing_protocol": audit.VERSION,
                   "started_utc": stamp(10), "dispatch_returned_utc": stamp(20),
                   "finished_utc": stamp(40), "window": deepcopy(EXPECTED),
                   "post_refresh": {"method": "sky.get_window_state", "success": True,
                       "started_utc": stamp(30), "finished_utc": stamp(40),
                       "window": deepcopy(EXPECTED), "screenshot_id": "provider-local-0"}}
        result.append((request, receipt))
    return result


class ProtocolBoundaries(unittest.TestCase):
    def reject(self, rows, code, events=()):
        with self.assertRaisesRegex(audit.ProtocolError, f"^{code}$"):
            audit.audit_records(rows, EXPECTED, events)

    def test_valid_reused_screenshot_ids_and_closed_event_boundaries(self):
        rows = records()
        events = [{"event": "project_selected", "utc": rows[0][1]["started_utc"]},
                  {"event": "theme_selected", "utc": rows[1][1]["finished_utc"]}]
        result = audit.audit_records(rows, EXPECTED, events)
        self.assertEqual(result["business_transition_count"], 2)
        self.assertEqual(result["operation_count"], 2)

    def test_wrong_input_window(self):
        rows = records()
        rows[0][1]["window"]["id"] += 1
        self.reject(rows, "input_window_differs_from_claim")

    def test_wrong_refresh_window(self):
        rows = records()
        rows[0][1]["post_refresh"]["window"]["title"] = "Other"
        self.reject(rows, "post_refresh_changed_window")

    def test_missing_refresh(self):
        rows = records()
        del rows[0][1]["post_refresh"]
        self.reject(rows, "missing_fresh_post_refresh")

    def test_refresh_failure_not_business_success(self):
        for value in (False, None, 1):
            with self.subTest(value=value):
                rows = records()
                rows[0][1]["post_refresh"]["success"] = value
                self.reject(rows, "post_refresh_method_or_success_missing")

    def test_refresh_error_and_missing_screenshot(self):
        rows = records()
        rows[0][1]["post_refresh"]["error"] = "fresh capture failed"
        self.reject(rows, "post_refresh_reported_error")
        rows = records()
        rows[0][1]["post_refresh"]["screenshot_id"] = " "
        self.reject(rows, "post_refresh_missing_screenshot")

    def test_reverse_timestamp_order(self):
        rows = records()
        rows[0][1]["dispatch_returned_utc"] = "2026-10-29T00:00:01.050Z"
        self.reject(rows, "operation_timestamp_order_invalid")

    def test_archival_finish_cannot_replace_refresh_finish(self):
        rows = records()
        rows[0][1]["finished_utc"] = "2026-10-29T00:00:01.090Z"
        self.reject(rows, "operation_finish_not_original_refresh_finish_string")

    def test_equivalent_instant_differently_serialized_is_not_exact_captured_string(self):
        rows = records()
        rows[0][1]["finished_utc"] = "2026-10-29T00:00:01.040+00:00"
        self.reject(rows, "operation_finish_not_original_refresh_finish_string")

    def test_overlapping_or_touching_closed_intervals(self):
        for start in ("2026-10-29T00:00:01.039Z", "2026-10-29T00:00:01.040Z"):
            with self.subTest(start=start):
                rows = records()
                rows[1][1]["started_utc"] = start
                rows[1][0].pop("created_utc")
                self.reject(rows, "operation_intervals_overlap_or_touch")

    def test_request_action_and_title_are_not_rewritten(self):
        rows = records()
        rows[0][1]["action"]["click_count"] = 2
        self.reject(rows, "raw_native_action_changed")
        rows = records()
        rows[0][0]["required_window_title"] = "Other"
        self.reject(rows, "request_target_window_changed")

    def test_duplicate_input_id(self):
        rows = records()
        rows[1][0]["id"] = rows[0][0]["id"]
        rows[1][1]["id"] = rows[0][0]["id"]
        self.reject(rows, "duplicate_or_empty_input_id")

    def test_outside_event_not_tolerated(self):
        self.reject(records(), "business_event_missing_or_ambiguous_interval",
                    [{"event": "project_selected", "utc": "2026-10-29T00:00:01.040001Z"}])

    def test_multiple_business_events_cannot_share_an_input(self):
        self.reject(records(), "one_input_claimed_multiple_business_transitions",
                    [{"event": "project_selected", "utc": "2026-10-29T00:00:01.020Z"},
                     {"event": "theme_selected", "utc": "2026-10-29T00:00:01.030Z"}])

    def test_owner_boundary_and_utc_offset(self):
        with self.assertRaisesRegex(audit.ProtocolError, "^operation_outside_actual_owner_lifetime$"):
            audit.audit_records(records(), EXPECTED,
                                owner_interval=("2026-10-29T00:00:02Z", "2026-10-29T00:00:04Z"))
        rows = records()
        rows[0][1]["started_utc"] = "2026-10-29T00:00:01.010+01:00"
        self.reject(rows, "timestamp_not_explicit_utc")


if __name__ == "__main__":
    unittest.main(verbosity=2)
