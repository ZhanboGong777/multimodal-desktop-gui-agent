"""Install one declared test-only validator; no model, capture or input calls."""
from __future__ import annotations

import hashlib
import importlib.metadata
import json
import os
import shutil
import subprocess
import time
from datetime import UTC, datetime
from pathlib import Path

import psutil

REPO = Path('C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree')
BASE = REPO / 'Document/Week6/results/full_requirements_20261028/target_bindings_001'
PYTHON = 'D:/Developer/multimodal-desktop-gui-agent/.venv-week5/Scripts/python.exe'


def write(path, value):
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write('\n')


def utc():
    return datetime.now(UTC).isoformat()


def versions():
    selected = ('jsonschema', 'attrs', 'referencing', 'rpds-py', 'jsonschema-specifications',
                'pydantic', 'numpy', 'Pillow', 'pytesseract', 'httpx', 'openai', 'torch')
    result = {}
    for name in selected:
        try:
            result[name] = importlib.metadata.version(name)
        except importlib.metadata.PackageNotFoundError:
            result[name] = None
    return result


def identity(process):
    return {'pid': process.pid, 'create_time': process.create_time(),
            'exe': process.exe(), 'cmdline': process.cmdline()}


def main():
    directory = BASE / 'schema_test_dependency_001'
    directory.mkdir(parents=True, exist_ok=False)
    dependency_file = REPO / 'requirements-dev.txt'
    shutil.copyfile(dependency_file, directory / 'requirements-dev.before.txt')
    shutil.copyfile(__file__, directory / 'owner_harness.py.txt')
    raw = dependency_file.read_bytes()
    assert b'jsonschema' not in raw
    before = versions()
    write(directory / 'before.json', {'utc': utc(), 'versions': before,
          'requirements_dev_sha256': hashlib.sha256(raw).hexdigest()})
    dependency_file.write_text(raw.decode('utf-8').rstrip() + '\njsonschema>=4.18,<5\n',
                               encoding='utf-8', newline='\n')
    command = [PYTHON, '-X', 'utf8', '-m', 'pip', 'install', 'jsonschema>=4.18,<5',
               '--disable-pip-version-check', '--no-input']
    started, tick, seen = utc(), time.monotonic(), {}
    write(directory / 'owner_claim.json', {'owner': identity(psutil.Process()),
          'command': command, 'started_utc': started, 'dependency_scope': 'test-only'})
    env = os.environ.copy()
    env.update(PYTHONUTF8='1', PYTHONIOENCODING='utf-8')
    with (directory / 'pip.stdout.txt').open('x', encoding='utf-8') as out:
        with (directory / 'pip.stderr.txt').open('x', encoding='utf-8') as err:
            child = subprocess.Popen(command, cwd=REPO, env=env, stdout=out, stderr=err)
            launcher = identity(psutil.Process(child.pid))
            while child.poll() is None:
                try:
                    for process in psutil.Process(child.pid).children(recursive=True):
                        key = (process.pid, process.create_time())
                        if key not in seen:
                            seen[key] = identity(process)
                except (psutil.NoSuchProcess, psutil.AccessDenied):
                    pass
                time.sleep(.25)
            code = child.wait()
    after = versions()
    unchanged_runtime = all(before[name] == after[name] for name in
                            ('pydantic', 'numpy', 'Pillow', 'pytesseract', 'httpx', 'openai', 'torch'))
    result = {'schema': 'week6.schema_test_dependency.actual.v1', 'started_utc': started,
              'finished_utc': utc(), 'elapsed_seconds': time.monotonic() - tick,
              'command': command, 'launcher': launcher, 'observed_descendants': list(seen.values()),
              'actual_child_exit_code': code, 'before_versions': before, 'after_versions': after,
              'requirements_dev_sha256': hashlib.sha256(dependency_file.read_bytes()).hexdigest(),
              'runtime_selected_versions_unchanged': unchanged_runtime,
              'checks_passed': code == 0 and after['jsonschema'] is not None and unchanged_runtime,
              'model_gui_calls': 0, 'runtime_dependency_added': False}
    write(directory / 'receipt.json', result)
    print(json.dumps(result))
    return 0 if result['checks_passed'] else 1


if __name__ == '__main__':
    raise SystemExit(main())
