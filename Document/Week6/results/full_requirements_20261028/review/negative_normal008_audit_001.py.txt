def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def unique(pairs):
    result = {}
    for key, value in pairs:
        assert key not in result, ("duplicate_json_key", key)
        result[key] = value
    return result


def bind(path, external=False):
    path = path.resolve()
    assert path.is_file() and (external or path.is_relative_to(BATCH.resolve())), path
    digest = sha(path)
    key = str(path) if external else path.relative_to(BATCH).as_posix()
    assert key not in FILES or FILES[key] == digest, ("changed_evidence", path)
    FILES[key] = digest
    return path


def load(path, external=False):
    return json.loads(bind(path, external).read_bytes(), object_pairs_hook=unique)


def stamp(value):
    result = datetime.fromisoformat(value)
    assert result.tzinfo
    return result
