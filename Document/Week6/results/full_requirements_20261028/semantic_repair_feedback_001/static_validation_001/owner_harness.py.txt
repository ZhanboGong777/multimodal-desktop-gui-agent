from pathlib import Path
from datetime import datetime, timezone
import ast
import copy
import hashlib
import json
import os
import subprocess
import time

root = Path(r"C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree")
base = root / "Document/Week6/results/full_requirements_20261028/semantic_repair_feedback_001"
out = base / "static_validation_001"
out.mkdir(exist_ok=False)
(out / "owner_harness.py.txt").write_bytes(Path(__file__).read_bytes())
keys = ["src/gui_agent/planning/semantic_workflow.py", "tests/test_week6_semantic_repair_feedback.py"]
sha = lambda raw: hashlib.sha256(raw).hexdigest()
before = {key: sha((root / key).read_bytes()) for key in keys}
for key in keys:
    ast.parse((root / key).read_text("utf-8"), filename=key)
old = ast.parse((base / "source_before/src/gui_agent/planning/semantic_workflow.py").read_text("utf-8"))
new = ast.parse((root / keys[0]).read_text("utf-8"))
names = ["_caller_contract_issue", "_validate_native_predicates", "_semantic_response_format", "SemanticGoal", "SemanticPlan", "SemanticPlanResult"]
equal = {}
for name in names:
    a = next(node for node in old.body if isinstance(node, (ast.ClassDef, ast.FunctionDef)) and node.name == name)
    b = next(node for node in new.body if isinstance(node, (ast.ClassDef, ast.FunctionDef)) and node.name == name)
    equal[name] = ast.dump(a) == ast.dump(b)
assert all(equal.values())
def validator(tree):
    cls = next(node for node in tree.body if isinstance(node, ast.ClassDef) and node.name == "SemanticDecomposer")
    return next(node for node in cls.body if isinstance(node, ast.FunctionDef) and node.name == "_validate")
class EraseTypedException(ast.NodeTransformer):
    def visit_Name(self, node):
        if node.id == "_DuplicateSemanticPostconditionsError":
            return ast.copy_location(ast.Name(id="ValueError", ctx=node.ctx), node)
        return node
normalized = EraseTypedException().visit(copy.deepcopy(validator(new)))
assert ast.dump(validator(old)) == ast.dump(normalized)
old_base = next(node.value for node in old.body if isinstance(node, ast.Assign)
    and any(isinstance(target, ast.Name) and target.id == "SEMANTIC_SYSTEM_PROMPT" for target in node.targets))
new_base = next(node.value for node in new.body if isinstance(node, ast.Assign)
    and any(isinstance(target, ast.Name) and target.id == "SEMANTIC_SYSTEM_PROMPT" for target in node.targets))
assert ast.literal_eval(old_base) == ast.literal_eval(new_base)
command = [r"D:/Developer/multimodal-desktop-gui-agent/.venv-week5/Scripts/python.exe", "-m", "ruff", "check", *keys]
started_at = datetime.now(timezone.utc).isoformat()
started = time.monotonic()
with (out / "ruff.stdout.txt").open("xb") as stdout, (out / "ruff.stderr.txt").open("xb") as stderr:
    child = subprocess.Popen(command, cwd=root, stdout=stdout, stderr=stderr, env={**os.environ, "PYTHONUTF8": "1"})
    code = child.wait()
after = {key: sha((root / key).read_bytes()) for key in keys}
payload = {"schema": "week6.semantic_repair_feedback.static_validation.v1", "command": command,
    "launcher_pid": child.pid, "started_at_utc": started_at, "elapsed_seconds": time.monotonic() - started,
    "actual_wait_exit_code": code, "owned_pins_before": before, "owned_pins_after": after,
    "owned_pins_unchanged": before == after, "ast_syntax_valid": True,
    "unchanged_native_and_preflight_AST": equal,
    "validator_AST_identical_except_exception_subclass": True,
    "base_system_constant_identical": True, "pytest_or_collection_executed": False,
    "no_actual_model_gui_calls": True, "not_combined_or_full_validation": True}
with (out / "validation.json").open("x", encoding="utf-8", newline="\n") as handle:
    json.dump(payload, handle, ensure_ascii=False, indent=2)
    handle.write("\n")
print(json.dumps({"path": str(out / "validation.json"), "sha256": sha((out / "validation.json").read_bytes()),
    "ruff_wait_exit": code, "owned_pins_unchanged": before == after, "source_pins": after}, ensure_ascii=False))
raise SystemExit(0 if code == 0 and before == after else 1)
