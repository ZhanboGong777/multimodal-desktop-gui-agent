def frame(identifier,texts):
    return ObservationSnapshot(observation_id=identifier,captured_at=datetime.now(UTC),
        screen_info=ScreenInfo(screenshot_width=800,screenshot_height=600,
                               control_width=800,control_height=600),
        elements=[ElementRef(element_id=f'{identifier}-e{index:03d}',text=text,
                    bounding_box=BoundingBox(left=20,top=20+index*60,right=400,bottom=60+index*60),
                    center=BoundingBox(left=20,top=20+index*60,right=400,bottom=60+index*60).center,
                    confidence=.99)
                  for index,text in enumerate(texts)])

def evaluate(name,expected,forbidden):
    task=TaskSpec(case_id='CHECK',instruction='Check proposed local progress only',
                  expect_text=expected,forbid_text=forbidden)
    proposal=DecompositionProposal.model_validate({'status':'ready','reason':'','subgoals':[{
        'instruction':'Advance the current local stage','expect_text':expected,'forbid_text':forbidden}]})
    error=''
    try:
        _check_proposal_constraints(proposal,request,initial,[])
        allowed=True
    except ValueError as exc:
        allowed=False
        error=str(exc)
    return {'case':name,'expect_text':expected,'forbid_text':forbidden,
            'initial_full_check':verifier.check_task(task,initial).model_dump(mode='json'),
            'following_full_check':verifier.check_task(task,following).model_dump(mode='json'),
            'current_checker_allowed':allowed,'current_checker_error':error}
