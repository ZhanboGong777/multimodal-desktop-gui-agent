def canon(node):
    return ast.dump(node, include_attributes=False)


def tree(path):
    return ast.parse(path.read_text('utf-8'))


def sha(path):
    return sha256(path.read_bytes()).hexdigest()


def fn(module, name):
    return next(n for n in module.body if isinstance(n, ast.FunctionDef) and n.name == name)


def assigned(node, name):
    return isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == name for t in node.targets)


def remove_import(module, name):
    count = 0
    for node in list(module.body):
        if isinstance(node, (ast.Import, ast.ImportFrom)):
            kept = [a for a in node.names if a.name != name]
            count += len(node.names)-len(kept)
            node.names = kept
            if not kept:
                module.body.remove(node)
    assert count == 1, (name, count)


class RemoveQueryMetadata(ast.NodeTransformer):
    def __init__(self, remove_assignment=False):
        self.fields = 0
        self.assignments = 0
        self.remove_assignment = remove_assignment

    def visit_Assign(self, node):
        if self.remove_assignment and assigned(node, 'query_contract'):
            assert canon(node.value) == canon(ast.parse('observed_query_contract(binding_data)', mode='eval').body)
            self.assignments += 1
            return None
        return self.generic_visit(node)

    def visit_Dict(self, node):
        self.generic_visit(node)
        pairs = []
        for key, value in zip(node.keys, node.values):
            if isinstance(key, ast.Constant) and key.value in {'observed_query_contract','observed_query_contract_sha256'}:
                self.fields += 1
            else:
                pairs.append((key,value))
        node.keys = [p[0] for p in pairs]
        node.values = [p[1] for p in pairs]
        return node
