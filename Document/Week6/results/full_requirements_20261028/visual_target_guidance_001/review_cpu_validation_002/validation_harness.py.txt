"""Actual offline owned-node execution; unique immutable raw receipts."""
import argparse
import hashlib
import json
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path("C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree")
BASE = ROOT / "Document/Week6/results/full_requirements_20261028/visual_target_guidance_001"
PYTHON = Path("D:/Developer/multimodal-desktop-gui-agent/.venv-week5/Scripts/python.exe")
OWNED = ["scripts/week6_audit_semantic_acceptance.py", "tests/test_week6_visual_target_guidance_review.py"]

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def pins():
    names = set(json.loads((BASE.parent / "target_bindings_001/final_source_freeze_001.json").read_bytes())["code_pins"])
    names.add(OWNED[1])
    return {name: sha(ROOT / name) for name in sorted(names)}

args = argparse.ArgumentParser()
args.add_argument("--name", required=True)
args = args.parse_args()
assert Path(args.name).name == args.name
out = BASE / args.name
out.mkdir(exist_ok=False)
before = pins()
for name in OWNED:
    dst = out / "source_before" / (name + ".txt")
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_bytes((ROOT / name).read_bytes())
(out / "pins_before.json").write_text(json.dumps(before, indent=2) + "\n", encoding="utf8")
commands = [
    ("ruff", [str(PYTHON), "-m", "ruff", "check", *OWNED]),
    ("pytest", [str(PYTHON), "-m", "pytest", "-q", OWNED[1], "--tb=short", "--basetemp", str(out / "pytest_tmp")]),
]
receipts = []
for label, command in commands:
    started = datetime.now(timezone.utc).isoformat()
    clock = time.monotonic()
    with (out / (label + ".stdout.txt")).open("wb") as stdout, (out / (label + ".stderr.txt")).open("wb") as stderr:
        owner = subprocess.Popen(command, cwd=ROOT, stdout=stdout, stderr=stderr)
        actual_wait = owner.wait()
    row = {"command": command, "cwd": str(ROOT), "pid": owner.pid, "actual_wait": actual_wait,
        "started_utc": started, "finished_utc": datetime.now(timezone.utc).isoformat(),
        "elapsed_seconds": time.monotonic() - clock,
        "stdout_sha256": sha(out / (label + ".stdout.txt")), "stderr_sha256": sha(out / (label + ".stderr.txt"))}
    receipts.append(row)
    (out / (label + ".receipt.json")).write_text(json.dumps(row, ensure_ascii=False, indent=2) + "\n", encoding="utf8")
after = pins()
(out / "pins_after.json").write_text(json.dumps(after, indent=2) + "\n", encoding="utf8")
validation = {"schema": "week6.visual_guidance.independent_offline_validation.v1",
    "scope": "owned new synthetic review tests and Ruff only; not shared/full regression, model or GUI evidence",
    "commands": receipts, "pin_count": len(before), "source_unchanged": before == after,
    "owned_sha256": {name: before[name] for name in OWNED},
    "before_after_changes": [name for name in before if before[name] != after[name]],
    "passed": all(row["actual_wait"] == 0 for row in receipts) and before == after}
(out / "validation.json").write_text(json.dumps(validation, ensure_ascii=False, indent=2) + "\n", encoding="utf8")
(out / "validation_harness.py.txt").write_bytes(Path(__file__).read_bytes())
print(json.dumps({"directory": str(out), "validation_sha256": sha(out / "validation.json"), **validation}, ensure_ascii=True))
raise SystemExit(0 if validation["passed"] else 1)
