def load(path):
    return json.loads(path.read_text(encoding="utf8"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def bound(path):
    return {"path": str(path), "sha256": sha(path), "bytes": path.stat().st_size}

    assert sha(FREEZE) == FREEZE_SHA and sha(PREREG) == PREREG_SHA
    freeze, previous, prereg = load(FREEZE), load(PREVIOUS), load(PREREG)

    assert result == observed["automatic_result"] and result["status"] == "blocked"
    assert result["planning_calls"] == 2 and result["model_requests"] == 3
    assert result["total_actions"] == result["unconfirmed_action_attempts"] == 0
    assert result["semantic_goals_total"] == 2 and result["completed_subgoals"] == []
    assert result["final_verification"] is None and len(result["goal_outcomes"]) == 1
    assert observed["runtime_pass"] is observed["independent_pass"] is observed["attribution_pass"] is False
    assert observed["source_unchanged"] is True and observed["input_receipts"] == observed["transition_attribution"] == []

    assert auditor.ROOT == replay
    assert Path(sys.modules["gui_agent.planning.planner"].__file__).resolve() == replay / "src/gui_agent/planning/planner.py"

        content = row["response"]["content"]
        assert row["http_responses"][0]["body"]["choices"][0]["message"]["content"] == content
