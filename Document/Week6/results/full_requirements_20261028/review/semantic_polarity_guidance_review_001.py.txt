ALLOWED = {"SEMANTIC_SYSTEM_PROMPT", "SEMANTIC_GENERIC_REPAIR"}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def assignments(tree):
    return {node.targets[0].id: ast.literal_eval(node.value) for node in tree.body
            if isinstance(node, ast.Assign) and len(node.targets) == 1
            and isinstance(node.targets[0], ast.Name) and node.targets[0].id in ALLOWED}


def normalized(tree):
    result = deepcopy(tree)
    for node in result.body:
        if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
            if node.targets[0].id in ALLOWED:
                node.value = ast.Constant(value="__INDEPENDENT_REVIEWED_FIXED_PROMPT_CONSTANT__")
    return ast.dump(result, include_attributes=False)
