"""Own combined, post-edit CPU checks; never starts a model or desktop task."""
from __future__ import annotations

import argparse
import hashlib
import json
import os
import re
import shutil
import subprocess
import time
from datetime import UTC, datetime
from pathlib import Path

import psutil

REPO = Path('C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree')
BASE = REPO / 'Document/Week6/results/full_requirements_20261028'
PYTHON = 'D:/Developer/multimodal-desktop-gui-agent/.venv-week5/Scripts/python.exe'


def utc():
    return datetime.now(UTC).isoformat()


def write(path, value):
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write('\n')


def pins():
    selected = {REPO / 'pyproject.toml'}
    for name in ('src', 'tests', 'scripts'):
        selected.update((REPO / name).rglob('*.py'))
    selected.update(p for p in (REPO / 'configs').rglob('*') if p.is_file())
    return {p.relative_to(REPO).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in sorted(selected)}


def identity(process):
    return {'pid': process.pid, 'create_time': process.create_time(),
            'exe': process.exe(), 'cmdline': process.cmdline()}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--name', required=True)
    parser.add_argument('--changed-only', action='store_true')
    args = parser.parse_args()
    assert args.name and all(c in 'abcdefghijklmnopqrstuvwxyz0123456789_-' for c in args.name)
    directory = BASE / 'semantic_conversation_repair_001' / args.name
    directory.mkdir(exist_ok=False)
    shutil.copyfile(__file__, directory / 'owner_harness.py.txt')
    old = json.loads((BASE / 'semantic_predicted_postconditions_001/cpu_validation_001/validation.json').read_text(encoding='utf-8'))
    old_command = old['receipts'][1]['command']
    tests = [item for item in old_command if item.startswith('tests/') and item.endswith('.py')]
    tests += [
        'tests/test_week6_semantic_live_monitoring.py',
        'tests/test_week6_semantic_repair_feedback.py',
        'tests/test_week6_semantic_repair_feedback_review.py',
        'tests/test_week6_semantic_conversation_repair.py',
        'tests/test_week6_semantic_conversation_repair_review.py',
    ]
    tests = list(dict.fromkeys(tests))
    if args.changed_only:
        tests = [
            'tests/test_week6_semantic_goal_grouping.py',
            'tests/test_week6_semantic_repair_feedback.py',
            'tests/test_week6_semantic_repair_feedback_review.py',
            'tests/test_week6_semantic_conversation_repair.py',
            'tests/test_week6_semantic_conversation_repair_review.py',
        ]
    assert all((REPO / item).is_file() for item in tests)
    before = pins()
    write(directory / 'tested_code_pins_before.json', before)
    started, tick = utc(), time.monotonic()
    commands = [
        ('ruff', [PYTHON, '-m', 'ruff', 'check', 'src', 'tests', 'scripts']),
        ('pytest', [PYTHON, '-X', 'utf8', '-m', 'pytest', '-q', *tests,
                    '--basetemp', str(directory / 'pytest_tmp')]),
        ('collection', [PYTHON, '-X', 'utf8', '-m', 'pytest', '--collect-only', '-q']),
    ]
    write(directory / 'owner_claim.json', {'schema': 'week6.combined_target.claim.v1',
          'owner': identity(psutil.Process()), 'started_utc': started,
          'commands': commands, 'model_gui_calls': 0, 'full_regression': False})
    env = os.environ.copy()
    env.update(PYTHONUTF8='1', PYTHONIOENCODING='utf-8')
    receipts = []
    for name, command in commands:
        call_started, call_tick = utc(), time.monotonic()
        seen = {}
        with (directory / f'{name}.stdout.txt').open('x', encoding='utf-8') as out:
            with (directory / f'{name}.stderr.txt').open('x', encoding='utf-8') as err:
                child = subprocess.Popen(command, cwd=REPO, env=env, stdout=out, stderr=err)
                launcher = identity(psutil.Process(child.pid))
                while child.poll() is None:
                    try:
                        for process in psutil.Process(child.pid).children(recursive=True):
                            key = (process.pid, process.create_time())
                            if key not in seen:
                                seen[key] = identity(process)
                    except (psutil.NoSuchProcess, psutil.AccessDenied):
                        pass
                    time.sleep(.25)
                code = child.wait()
        receipt = {'name': name, 'command': command, 'launcher': launcher,
                   'observed_descendants': list(seen.values()), 'started_utc': call_started,
                   'finished_utc': utc(), 'elapsed_seconds': time.monotonic() - call_tick,
                   'actual_child_exit_code': code}
        write(directory / f'{name}.receipt.json', receipt)
        receipts.append(receipt)
    after = pins()
    write(directory / 'tested_code_pins_after.json', after)
    target_output = (directory / 'pytest.stdout.txt').read_text(encoding='utf-8')
    collection_output = (directory / 'collection.stdout.txt').read_text(encoding='utf-8')
    target = re.search(r'(\d+) passed in ([\d.]+)s', target_output)
    failures = re.search(r'(\d+) failed[, ]', target_output)
    collected = re.search(r'(\d+) tests? collected in ([\d.]+)s', collection_output)
    passed = before == after and all(item['actual_child_exit_code'] == 0 for item in receipts)
    passed = bool(passed and target and collected)
    value = {'schema': 'week6.combined_target.actual.v1', 'started_utc': started,
             'scope': 'changed_tests_only' if args.changed_only else 'affected_semantic',
             'finished_utc': utc(), 'elapsed_seconds': time.monotonic() - tick,
             'receipts': receipts, 'code_pin_count': len(before), 'code_unchanged': before == after,
             'targeted_tests_passed': int(target[1]) if target else None,
             'targeted_failed_count': int(failures[1]) if failures else 0,
             'targeted_pytest_seconds': float(target[2]) if target else None,
             'collected_test_count': int(collected[1]) if collected else None,
             'checks_passed': passed, 'model_gui_calls': 0, 'full_regression': False}
    write(directory / 'validation.json', value)
    print(json.dumps({k: v for k, v in value.items() if k != 'receipts'}))
    return 0 if passed else 1


if __name__ == '__main__':
    raise SystemExit(main())
