"""Narrow independent target with real waits and explicit before/after code pins."""
from __future__ import annotations

import hashlib
import json
import os
import subprocess
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

import psutil

ROOT = Path("C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree")
BASE = ROOT / "Document/Week6/results/full_requirements_20261028"
OUT = BASE / "target_bindings_001/review_cpu_validation_001"
NEW_FILES = ["src/gui_agent/planning/target_bindings.py", "tests/test_week6_target_bindings.py",
             "tests/test_week6_target_bindings_review.py"]


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def identity(process):
    return {"pid": process.pid, "create_time": process.create_time(), "exe": process.exe()}


def main():
    OUT.mkdir(exist_ok=False)
    prior = json.loads((BASE / "action_output_contract_001/final_source_freeze_001.json").read_text("utf8"))
    paths = sorted(set(prior["code_pins"]) | set(NEW_FILES))
    before = {name: sha(ROOT / name) for name in paths}
    assert len(before) == 270
    (OUT / "source_pins_before.json").write_text(json.dumps(before, indent=2) + "\n", encoding="utf8")
    for name in ["scripts/week6_audit_semantic_acceptance.py", "tests/test_week6_target_bindings_review.py", *NEW_FILES[:2],
                 "src/gui_agent/planning/planner.py", "src/gui_agent/runtime/runner.py"]:
        path = OUT / "source_before" / (name + ".txt")
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes((ROOT / name).read_bytes())
    receipts = []
    commands = [("pytest", [sys.executable, "-m", "pytest", "-q", NEW_FILES[2],
                             "--basetemp=" + str(OUT / "pytest_tmp"), "--tb=short"]),
                ("ruff", [sys.executable, "-m", "ruff", "check", "scripts/week6_audit_semantic_acceptance.py", NEW_FILES[2]])]
    for label, command in commands:
        started = datetime.now(timezone.utc).isoformat()
        timer = time.monotonic()
        observed = {}
        with (OUT / f"{label}.stdout.txt").open("xb") as stdout, (OUT / f"{label}.stderr.txt").open("xb") as stderr:
            child = subprocess.Popen(command, cwd=ROOT, stdout=stdout, stderr=stderr,
                                     creationflags=subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0)
            launcher = identity(psutil.Process(child.pid))
            while child.poll() is None:
                try:
                    for descendant in psutil.Process(child.pid).children(recursive=True):
                        try:
                            item = identity(descendant)
                            observed[(item["pid"], item["create_time"])] = item
                        except psutil.Error:
                            pass
                except psutil.Error:
                    pass
                time.sleep(.05)
            actual_exit = child.wait()
        receipt = {"command": command, "started_utc": started,
                   "finished_utc": datetime.now(timezone.utc).isoformat(), "elapsed_seconds": time.monotonic() - timer,
                   "launcher": launcher, "observed_descendants": list(observed.values()),
                   "actual_wait_exit_code": actual_exit, "stdout_sha256": sha(OUT / f"{label}.stdout.txt"),
                   "stderr_sha256": sha(OUT / f"{label}.stderr.txt")}
        (OUT / f"{label}.receipt.json").write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + "\n", encoding="utf8")
        receipts.append(receipt)
    after = {name: sha(ROOT / name) for name in paths}
    (OUT / "source_pins_after.json").write_text(json.dumps(after, indent=2) + "\n", encoding="utf8")
    result = {"schema": "week6.target_bindings.independent_review.actual.v1", "scope": "new_independent_review_test_only",
              "receipts": receipts, "code_pin_count": len(before), "code_pins_unchanged": before == after,
              "checks_passed": before == after and all(item["actual_wait_exit_code"] == 0 for item in receipts),
              "no_model_gui_or_experiment_rerun": True, "full_regression": False}
    (OUT / "validation.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf8")
    print(json.dumps({"validation": str(OUT / "validation.json"), "sha256": sha(OUT / "validation.json"),
                      "exits": [item["actual_wait_exit_code"] for item in receipts], "pins_same": before == after}, ensure_ascii=False))


if __name__ == "__main__":
    main()
