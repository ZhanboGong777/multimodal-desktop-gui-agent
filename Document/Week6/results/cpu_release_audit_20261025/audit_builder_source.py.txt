def read(path):
    return json.loads(path.read_bytes())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def pins():
    files = [path for name in ('src','tests','scripts','configs')
             for path in (REPO / name).rglob('*') if path.is_file() and path.suffix in ('.py','.json','.yaml')]
    files += [REPO/'pyproject.toml',REPO/'README.md']
    return {path.relative_to(REPO).as_posix():sha(path) for path in sorted(files)}

validation = read(VALIDATION/'validation.json')

new_paths = sorted(before_current.keys()-before_published.keys())
missing_paths = sorted(before_published.keys()-before_current.keys())

                       if before_published[path]!=before_current[path])

assert validation['changed_tested_sources']==[]
assert not new_paths and not missing_paths and not changed_paths

observation_path = OLD_GUI/'automatic/observation_round_003/observation.json'
observation = ObservationSnapshot.model_validate(read(observation_path))


def proposal(expected=(),forbidden=()):
    return DecompositionProposal.model_validate({'status':'ready','reason':'','subgoals':[{
        'instruction':'Advance the current local stage', 'expect_text':list(expected),
        'forbid_text':list(forbidden)}]})


cases = [
    ('actual_008_bad_intermediate_proposal', DecompositionProposal.model_validate({
        'status':'ready','reason':'','subgoals':[{key:goal[key] for key in
        ('instruction','expect_text','forbid_text')} for goal in old_proposal['subgoals']]}), False),
    ('case_insensitive_final_forbid_conflict',proposal(['Review complete'],['project alpha']),False),
    ('substring_final_forbid_conflict',proposal(['Review complete'],['alpha']),False),
    ('already_visible_operation_label',proposal(['Review report'],['Draft ready']),False),
    ('persistent_final_context_alone',proposal(['Project ALPHA']),False),
    ('persistent_context_plus_unmet_progress',proposal(['Project ALPHA','Review complete']),True),
    ('visible_stage_marker_must_disappear',proposal([],['Draft ready']),True),
]
probes=[]
for name,value,expected_allowed in cases:
    error=''
    try:
        _check_proposal_constraints(value,request,observation,[])
        allowed=True
    except ValueError as exc:
        allowed=False
        error=str(exc)
    probes.append({'case':name,'expected_allowed':expected_allowed,'allowed':allowed,
                   'error':error,'passed':allowed==expected_allowed})
assert all(row['passed'] for row in probes)
assert request.model_dump(mode='json')==original_final_rules
