from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json

root = Path(r"C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree")
base = root / "Document/Week6/results/full_requirements_20261028"
out = base / "semantic_repair_feedback_001"
out.mkdir(exist_ok=False)
(out / "archive_harness.py.txt").write_bytes(Path(__file__).read_bytes())
freeze_path = base / "grounding_consistency_001/final_source_freeze_001.json"
freeze_raw = freeze_path.read_bytes()
sha = lambda raw: hashlib.sha256(raw).hexdigest()
assert sha(freeze_raw) == "f4a7ab5cf6b6f87b694b1777b8074054b3659b46e18489589f6b22ce235ec077"
pins = json.loads(freeze_raw)["code_pins"]
copies = []
def copy(source, destination):
    raw = source.read_bytes()
    destination.parent.mkdir(parents=True, exist_ok=True)
    with destination.open("xb") as handle:
        handle.write(raw)
    assert destination.read_bytes() == raw
    copies.append({"source": str(source), "destination": str(destination.relative_to(out)), "bytes": len(raw), "sha256": sha(raw)})
for key, expected in pins.items():
    assert sha((root / key).read_bytes()) == expected
    copy(root / key, out / "source_before" / key)
copy(root / "README.md", out / "source_before/README.md")
copy(freeze_path, out / "freeze_before/final_source_freeze_001.json")
copy(base / "review/negative_normal011_audit_001.json", out / "negative_normal011_audit_001.json")
normal = base / "live/normal_011"
files = sorted(path for path in normal.rglob("*") if path.is_file())
assert len(files) == 128
for source in files:
    copy(source, out / "normal011_before" / source.relative_to(normal))
assert {key: sha((root / key).read_bytes()) for key in pins} == pins
payload = {"schema": "week6.semantic_repair_feedback.before_archive.v1", "created_at_utc": datetime.now(timezone.utc).isoformat(), "old_freeze_sha256": sha(freeze_raw), "old_code_pin_count": len(pins), "source_before_after_equal": True, "normal011_original_file_count": len(files), "copies": copies, "scope": "Typed duplicate-postcondition repair presentation only; v6 native schema/guards/caller/budgets remain unchanged. No prior failed run is reclassified.", "models_gui_calls": 0, "pytest_scratch_copied": False}
manifest = out / "source_before_manifest.json"
with manifest.open("x", encoding="utf8", newline="\n") as handle:
    json.dump(payload, handle, ensure_ascii=False, indent=2)
    handle.write("\n")
print(json.dumps({"path": str(manifest), "sha256": sha(manifest.read_bytes()), "file_count": len(copies)}, ensure_ascii=False))
