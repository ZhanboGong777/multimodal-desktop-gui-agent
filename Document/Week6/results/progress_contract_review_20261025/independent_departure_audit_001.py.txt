    def __init__(self, content, *, supported=True, reject=False):
        super().__init__(model_name="CPU-no-model", max_retries=0)
        self.content, self.calls, self.reject = content, [], reject
        self.supports_strict_json_schema = supported

        if self.reject:
            raise b.ModelError("CPU simulated schema endpoint rejection")
        return b.ModelResponse(content=self.content, model_name=self.model_name, provider=self.name)
