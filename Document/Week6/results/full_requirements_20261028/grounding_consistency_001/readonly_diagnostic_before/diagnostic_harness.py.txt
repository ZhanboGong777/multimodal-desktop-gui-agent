def digest(data):
    return hashlib.sha256(data).hexdigest()

def write_json(name, payload):
    with (out / name).open("x", encoding="utf-8", newline="\n") as handle:
        json.dump(payload, handle, ensure_ascii=False, indent=2)
        handle.write("\n")
