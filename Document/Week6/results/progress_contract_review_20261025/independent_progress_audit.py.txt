def frame(name, texts, **changes):
    value = s.ObservationSnapshot(
        observation_id=name, captured_at=stamp + timedelta(seconds=name != "before"),
        screen_info=common.ScreenInfo(screenshot_width=800, screenshot_height=600,
                                     control_width=800, control_height=600),
        window_title="CPU hypothetical application", window_id="cpu-window-1",
        foreground_stable=True, window_bounds=common.BoundingBox(left=0, top=0, right=800, bottom=600),
        elements=[s.ElementRef(element_id=f"{name}-{i}", text=text,
                              bounding_box=common.BoundingBox(left=10, top=10+i*30, right=300, bottom=30+i*30),
                              center=common.Point(x=155, y=20+i*30), confidence=1)
                  for i, text in enumerate(texts)],
    )
    return value.model_copy(update=changes)


def prepare(root):
    root.mkdir()
    request = m.WorkflowRequest(instruction="Advance and finish the offline local task", expect_text=["Caller finished"])
    raw = WorkflowSpec(workflow_id="CPU", instruction=request.instruction,
                       subgoals=[s.TaskSpec(case_id="G01", instruction="Advance the local stage",
                                           expect_text=["Advance control"],
                                           forbid_text=["Pending stage", "Forbidden error"])],
                       final_check=s.TaskSpec(case_id="FINAL", instruction="Read only", expect_text=request.expect_text))
    before = frame("before", ["Advance control", "Pending stage"])
    derived, contract = m._derive_progress_contract(raw, request, before, root / "contract.json")
    return raw, derived, before, contract


def run_case(root, *, post=None, status="succeeded", unknown=0, steps=True,
             action_type="click", wrong_id=False, audit_error=None):
    raw, derived, before, contract = prepare(root)
    post = post or frame("post", ["Next stage"])
    frozen = (root / "contract.json").read_bytes()
    class Child:
        observer = SimpleNamespace(observe=lambda: post)
        verifier = Verifier()
        recorder = SimpleNamespace(directory=root)
        def run(self, task, options):
            assert (root / "contract.json").read_bytes() == frozen
            observed = self.observer.observe()
            result = self.verifier.check_task(task, observed)
            if wrong_id:
                result.evidence["observation_id"] = "never-observed"
            return s.TaskRunResult(
                run_id="CPU", case_id=task.case_id, instruction=task.instruction, execute=True,
                status=status, planning_calls=3, unconfirmed_action_attempts=unknown, verification=result,
                steps=[s.StepRecord(index=1, step_id="S1", description="CPU fake receipt", action_type=action_type,
                                    action_result={"success": True, "dry_run": False})] if steps else [],
            )
    audits = []
    original = m._write_new_json
    if audit_error:
        def deny(path, value):
            if path.name == "progress_verifications.json":
                raise audit_error("CPU injected audit persistence failure")
            return original(path, value)
        m._write_new_json = deny
    try:
        outcome = m._ProgressChild(Child(), raw.subgoals[0], derived.subgoals[0], contract, audits, 1).run(
            derived.subgoals[0], s.ExecutionOptions(execute=True))
    finally:
        m._write_new_json = original
    assert (root / "contract.json").read_bytes() == frozen
    return outcome, audits
