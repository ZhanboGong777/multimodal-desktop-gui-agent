"""Read-only ended-run audit; no model, GUI, tests, or production mutations."""

from __future__ import annotations

import ast
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path
from types import SimpleNamespace


ROOT = Path("C:/Users/联想/Documents/ChatGPT/Intern/week6-worktree")
BASE = ROOT / "Document/Week6/results/full_requirements_20261028"
CONTAINER = BASE / "live/normal_011"
RUN = CONTAINER / "run"
ROUND = RUN / "automatic/semantic_plan_round_001"
FREEZE = BASE / "grounding_consistency_001/final_source_freeze_001.json"
FREEZE_SHA = "f4a7ab5cf6b6f87b694b1777b8074054b3659b46e18489589f6b22ce235ec077"
DEST = BASE / "review/negative_normal011_audit_001.json"


def load(path):
    return json.loads(path.read_text(encoding="utf-8"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def bound(path):
    return {"path": str(path), "sha256": sha(path), "bytes": path.stat().st_size}


def new(path, content):
    with path.open("xb") as stream:
        stream.write(content)


def signature(goal):
    return (goal.instruction.casefold(), tuple(sorted(x.casefold() for x in goal.expect_text)),
            tuple(sorted(x.casefold() for x in goal.forbid_text)))


def main():
    assert not DEST.exists(), "Never overwrite an audit"
    assert sha(FREEZE) == FREEZE_SHA
    freeze, contract = load(FREEZE), load(RUN / "contract.json")
    owner, start = load(CONTAINER / "owner_receipt.json"), load(CONTAINER / "owner_start.json")
    runtime = load(RUN / "runtime_result.json")
    observed = load(RUN / "acceptance_observed.json")
    protocol, result = load(ROUND / "protocol.json"), load(ROUND / "result.json")
    monitor = load(RUN / "automatic/run_status.json")
    transport = [json.loads(s) for s in (RUN / "automatic/transport_requests.jsonl").read_text(encoding="utf-8").splitlines()]
    pins = freeze["code_pins"]
    assert len(pins) == freeze["code_pin_count"] == 261
    assert owner["child_exit_code"] == 1 and abs(owner["elapsed_seconds"] - 93.813) < .001
    assert owner["owner_harness_sha256"] == sha(CONTAINER / "owner_harness.py.txt")
    assert owner["command"] == start["command"]

    for relative, expected in contract["source_sha256"].items():
        snapshot = RUN / "source_snapshot" / (relative + ".txt")
        assert sha(snapshot) == expected == pins[relative]
        source_evidence.append({"relative_path": relative, "snapshot": bound(snapshot), "freeze_sha256": expected})
    current_differences = [relative for relative, expected in pins.items() if sha(ROOT / relative) != expected]
    assert not current_differences, current_differences
    assert runtime == observed["automatic_result"]
    assert runtime["status"] == "failed" and observed["runtime_pass"] is False
    assert runtime["total_actions"] == runtime["unconfirmed_action_attempts"] == 0
    assert runtime["planning_calls"] == runtime["model_requests"] == 3
    assert runtime["semantic_goals_total"] == 0 and not runtime["goal_outcomes"]
    assert not runtime["completed_subgoals"] and runtime["workflow_result"] is None
    assert runtime["final_verification"] is None
    assert observed["source_unchanged"] is True and not observed["input_receipts"]
    assert not observed["transition_attribution"]
    assert [e["event"] for e in observed["independent_events"]] == ["started"]
    assert monitor["actions_used"] == 0 and monitor["planning_pass"] == 3
    assert monitor["task_status"] == "failed" and monitor["subgoals_completed"] == monitor["subgoals_total"] == 0
    assert len(transport) == 3 and sum(t["requests_made"] for t in transport) == 3
    assert protocol["schema_version"] == "week6.semantic_subtasks.v6"
    assert protocol["postcondition_source"] == "untrusted_model_prediction"
    assert protocol["input_modality"] == "caller_intent_only" and protocol["snapshot_sent_to_model"] is False
    item_schema = protocol["response_format"]["json_schema"]["schema"]["$defs"]["SemanticGoal"]["properties"]["expect_text"]["items"]
    assert item_schema["type"] == "string" and "enum" not in item_schema
    assert not (ROUND / "native_proposal.json").exists() and not (ROUND / "compiled_workflow.json").exists()

    # Replay the exact pinned local validation method in an isolated namespace.
    # It throws before consulting the unused before frame; no GUI or model call.
    semantic_source = RUN / "source_snapshot/src/gui_agent/planning/semantic_workflow.py.txt"
    tree = ast.parse(semantic_source.read_text(encoding="utf-8"))
    cls = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == "SemanticDecomposer")
    method = next(n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name == "_validate")
    namespace = {"_signature": signature}
    exec(compile(ast.Module(body=[method], type_ignores=[]), str(semantic_source), "exec"), namespace)
    attempts, content_hashes = [], []
    previous_content = None
    for number, dispatch in enumerate(transport, 1):
        request_path = ROUND / f"attempt_{number:02d}_request.json"
        response_path = ROUND / f"attempt_{number:02d}_response.json"
        error_path = ROUND / f"attempt_{number:02d}_error.json"
        request, response, error = load(request_path), load(response_path), load(error_path)
        native = json.loads(response["content"])
        assert response["error"] is None and native["status"] == "ready"
        assert len(native["subgoals"]) == 3 and len(response["content"]) == 832
        assert request["instruction"] == contract["request"]["instruction"]
        assert request["context"]["original_caller_request"] == contract["request"]
        assert request["response_format"] == protocol["response_format"]
        assert request["image_path"] is None and request["image_attached"] is False
        assert "current_observation" not in request["context"]
        system_schema = request["system"].split("Exact response JSON schema (also supplied as response_format):\n", 1)[1]
        assert json.loads(system_schema) == protocol["response_format"]["json_schema"]["schema"]
        if number > 1:
            assert request["context"]["constraint_error"] == runtime["error"]
            assert request["context"]["previous_response"] == previous_content
            assert request["context"]["previous_response_characters"] == len(previous_content)
            assert Path(request["context"]["previous_response_record"]).resolve() == (ROUND / f"attempt_{number-1:02d}_response.json").resolve()
        assert dispatch["count_source"] == "httpx_request_event" and dispatch["requests_made"] == 1
        assert dispatch["error"] == "" and dispatch["response"]["content"] == response["content"]
        assert dispatch["messages"][0] == {"role": "system", "content": request["system"]}
        user = json.loads(dispatch["messages"][1]["content"])
        assert user == {"instruction": request["instruction"], "context": request["context"]}
        assert all(x["status_code"] == 200 for x in dispatch["http_responses"])
        assert dispatch["http_responses"][-1]["body"]["choices"][0]["message"]["content"] == response["content"]
        goals = [SimpleNamespace(**g) for g in native["subgoals"]]
        assert len({g.goal_id.casefold() for g in goals}) == 3
        assert len({g.instruction.casefold() for g in goals}) == 3
        for goal in goals:
            assert type(goal.instruction) is str and 0 < len(goal.instruction) <= 600 and goal.instruction == goal.instruction.strip()
            for values in (goal.expect_text, goal.forbid_text):
                assert type(values) is list and len(values) <= 8
                assert all(type(v) is str and 0 < len(v) <= 200 and v == v.strip() for v in values)
        assert signature(goals[1])[1:] == signature(goals[2])[1:]
        try:
            namespace["_validate"](SimpleNamespace(max_subgoals=8), SimpleNamespace(status="ready", subgoals=goals),
                                    SimpleNamespace(**contract["request"]), None, [], False)
        except ValueError as exc:
            reproduced = str(exc)
        else:
            raise AssertionError("duplicate checks unexpectedly passed")
        assert reproduced == error["error"] == runtime["error"]
        content_hash = hashlib.sha256(response["content"].encode()).hexdigest()
        content_hashes.append(content_hash)
        attempts.append({"attempt": number, "native": native, "response_content_sha256": content_hash,
                         "content_characters": len(response["content"]), "completion_tokens": response["usage"]["completion_tokens"],
                         "saved_response_latency_ms": response["latency_ms"], "http_status": 200,
                         "guard_error": error["error"], "isolated_exact_method_error_matches": True,
                         "repair_context_contains_previous_response_and_exact_error": number > 1})
        previous_content = response["content"]
    assert len(set(content_hashes)) == 1
    assert result["status"] == "failed" and result["planning_calls"] == result["model_requests"] == 3
    records = [bound(p) for p in sorted(CONTAINER.rglob("*")) if p.is_file()]
    script_copy = DEST.with_name("negative_normal011_audit_001.py.txt")
    new(script_copy, Path(__file__).read_bytes())
    audit = {
        "schema": "week6.independent.negative_semantic_live.v2",

        "reviewer": "/root/week6_replanning_review (Codex agent, not a human signature)",
        "scope": "Read-only ended normal_011 evidence plus isolated exact local validation replay; no model/GUI/owner/test rerun or production edits",
        "run": str(CONTAINER), "freeze": bound(FREEZE),
        "binding": {"current_all_261_pins_equal_freeze_at_review": True, "run_source_snapshot_count": len(source_evidence),
                    "run_source_snapshot_contract_and_freeze_match": True, "source_evidence": source_evidence},
        "actual_exit": {"child_exit_code": owner["child_exit_code"], "elapsed_seconds": owner["elapsed_seconds"],

                        "launcher": owner["launcher"], "observed_descendants": owner["observed_descendants"],
                        "harness_sha256_matches": True},
        "actual_accounting": {"planning_calls": 3, "http_requests_made": 3, "known_task_inputs": 0,
                              "unconfirmed_action_attempts": 0, "accepted_semantic_goals": 0,
                              "completed_goals": 0, "independent_events": ["started"],
                              "final_verification": None, "accounting_complete": runtime["accounting_complete"]},
        "native_attempts": attempts,
        "diagnosis": {
            "classification": "Native plan duplicate-postcondition constraint failure, before child creation/input",
            "precise_failure": "review_green_report and commit_green_report have the same complete expect_text [Project ALPHA, Theme GREEN, LOCAL COMPLETE] and forbid_text []; their different instructions/IDs do not provide distinct verifiable objectives.",
            "all_three_response_contents_identical": True,
            "not_a_refusal_or_user_parameter_clarification": True,
            "not_grounding_activation_or_telemetry_failure": True,
            "repair_error_and_full_previous_reply_reached_saved_second_and_third_call_arguments": True,
            "strict_transport_schema_does_not_force_caller_only_predicate_values": True,
            "logger_limit": "Saved messages are pre-wire call arguments; response bodies and httpx request-event counts are retained. This audit does not call them original outgoing wire request bodies.",
            "generic_code_correction_supported": False,
            "reason": "The exact frozen validator independently reproduces the saved error. The producer supplied explicit generic combine-or-distinguish repair guidance and the same native error recurred. No evidence here establishes a validator, accounting, omitted repair-context, or lingering caller-pool bug.",
            "possible_future_work_not_validated": "A separately versioned, bounded repair strategy may reduce echoing of the invalid prior answer, but this single failure does not establish that it will work. Preserve refusal, all semantic/post-input/final/event gates, and count any new experiment independently.",
            "forbidden_inferences": ["Delete duplicate-check guard", "Locally merge or rewrite model goals", "Treat ready as accepted", "Claim predicted review/commit outcome exists on the GUI", "Reuse zero-input failure as GUI completion", "Conclude the task is logically impossible"]
        },
        "artifacts": records,
        "audit_script": {"private": bound(Path(__file__)), "public_copy": bound(script_copy)},
    }
    new(DEST, (json.dumps(audit, ensure_ascii=False, indent=2) + "\n").encode("utf-8"))
    print(json.dumps({"audit": str(DEST), "sha256": sha(DEST), "bound_original_files": len(records),
                      "source_snapshots": len(source_evidence), "native_content_sha256": content_hashes[0]}, ensure_ascii=False))

    main()
