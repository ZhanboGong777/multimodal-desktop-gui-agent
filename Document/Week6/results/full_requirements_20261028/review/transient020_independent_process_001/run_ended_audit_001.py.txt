def load(path):
    return json.loads(path.read_text("utf-8"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def bound(path):
    return {"path": str(path), "sha256": sha(path), "bytes": path.stat().st_size}


def utc(value):
    result = datetime.fromisoformat(value)
    assert result.tzinfo is not None and result.utcoffset() == timedelta(0)
    return result


def canonical(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False)


def inventory(directory):
    return {path.relative_to(directory).as_posix(): sha(path)
            for path in directory.rglob("*") if path.is_file()}


def identity(process):
    return {"pid": process.pid, "create_time": process.create_time(), "exe": process.exe()}


def write(path, value):
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write("\n")


def audit_process(name, command):
    started, timer = datetime.now(timezone.utc).isoformat(), time.monotonic()
    descendants, launcher = {}, None
    with (OUT / f"{name}.stdout.txt").open("xb") as stdout, (OUT / f"{name}.stderr.txt").open("xb") as stderr:
        child = subprocess.Popen(command, cwd=ROOT, stdout=stdout, stderr=stderr,
            creationflags=subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0)
        try:
            launcher = identity(psutil.Process(child.pid))
        except psutil.Error:
            pass
        while child.poll() is None:
            try:
                for process in psutil.Process(child.pid).children(recursive=True):
                    try:
                        row = identity(process)
                        descendants[(row["pid"], row["create_time"])] = row
                    except psutil.Error:
                        pass
            except psutil.Error:
                pass
            time.sleep(.05)
        actual_wait = child.wait()
    receipt = {"schema": "week6.transient020.saved_evidence_audit.actual_exit.v1", "audit_kind": name,
        "command": command, "popen_pid": child.pid, "started_utc": started,
        "finished_utc": datetime.now(timezone.utc).isoformat(), "elapsed_seconds": time.monotonic() - timer,
        "launcher": launcher, "observed_descendants": list(descendants.values()),
        "actual_child_exit_code": actual_wait, "stdout": bound(OUT / f"{name}.stdout.txt"),
        "stderr": bound(OUT / f"{name}.stderr.txt")}
    write(OUT / f"{name}.receipt.json", receipt)
    return receipt

    assert handoff["exec_wait"]["exit_code"] == owner["child_exit_code"]
    assert canonical(handoff["owner_receipt"]) == canonical(owner)
    for row in [owner["owner"], owner["launcher"], *owner["observed_descendants"]]:
        try:
            assert psutil.Process(row["pid"]).create_time()!=row["create_time"], "Original task process still alive"
        except psutil.NoSuchProcess:
            pass

    assert type(handoff["exec_session"]) in (str, int) and str(handoff["exec_session"]).strip()
    command = owner["command"]

    assert command[command.index("--mode") + 1] == "transient"

    script = ROOT / "scripts/week6_audit_semantic_acceptance.py"

    # Pre-owner external Root acknowledgement anchors the original and case copy.
    assert sha(ACK) == args.claim_ack_sha256
